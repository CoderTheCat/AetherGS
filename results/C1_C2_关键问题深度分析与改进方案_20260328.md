```python
def compute_importance_scores(optimizer, gaussian_indices, grad_xyz, grad_opacity, grad_scale):
    """
    计算每个高斯点的重要性分数
    
    综合考虑：
    1. 梯度方差 - 反映学习活跃度
    2. 位置误差 - 反映重建质量
    3. 不透明度 - 反映可见性
    """
    # 1. 梯度方差分数 [N,]
    grad_var_xyz = torch.var(grad_xyz[gaussian_indices], dim=1)  # [N,]
    grad_var_opacity = torch.var(grad_opacity[gaussian_indices], dim=1)  # [N,]
    grad_var_scale = torch.var(grad_scale[gaussian_indices], dim=1)  # [N,]
    
    grad_score = grad_var_xyz.mean(dim=0) + grad_var_opacity + grad_var_scale.mean(dim=0)
    
    # 2. 归一化到 [0, 1]
    grad_score = (grad_score - grad_score.min()) / (grad_score.max() - grad_score.min() + 1e-8)
    
    # 3. 不透明度权重（可见的高斯点更重要）
    opacity = optimizer.param_groups[0]["params"][0][gaussian_indices].sigmoid()
    opacity_score = opacity.mean(dim=0)  # [N,]
    
    # 4. 综合分数
    importance_score = 0.7 * grad_score + 0.3 * opacity_score
    
    return importance_score
```

    """
    # 1. 获取需要分裂/克隆的候选点
    split_mask, clone_mask = self.get_split_clone_masks(optimizer, epoch)
    
    # 2. 计算预算
    budget = self.get_current_target(epoch) - self.current_points_num
    budget = max(0, budget)
    
    # 3. 计算重要性分数
    clone_indices = torch.where(clone_mask)[0]
    split_indices = torch.where(split_mask)[0]
    
    if len(clone_indices) > 0:
        clone_importance = compute_importance_scores(
            optimizer, clone_indices, grad_xyz, grad_opacity, grad_scale
        )
        
        # 4. 按重要性排序，选择 top-K
        selected_clone_count = min(int(budget * 0.5), len(clone_indices))
        _, top_indices = torch.topk(clone_importance, selected_clone_count)
        selected_clone_indices = clone_indices[top_indices]
        
        # 5. 更新 mask
        clone_mask = torch.zeros_like(clone_mask)
        clone_mask[selected_clone_indices] = True
    
    if len(split_indices) > 0:
        split_importance = compute_importance_scores(
            optimizer, split_indices, grad_xyz, grad_opacity, grad_scale
        )
        
        # 6. 按重要性排序，选择 top-K
        selected_split_count = min(int(budget * 0.5), len(split_indices))
        _, top_indices = torch.topk(split_importance, selected_split_count)
        selected_split_indices = split_indices[top_indices]
        
        # 7. 更新 mask
        split_mask = torch.zeros_like(split_mask)

class AdaptiveMarginController:
    def __init__(self, base_margin=0.1):
        self.base_margin = base_margin
        self.scene_complexity = None
    
    def estimate_scene_complexity(self, gaussians, cameras):
        """
        估计场景复杂度
        基于：高斯点密度、视角变化范围
        """
        # 1. 高斯点密度
        point_density = len(gaussians) / self.scene_volume
        
        # 2. 视角变化范围
        view_angle_range = self.compute_view_angle_range(cameras)
        
        # 3. 综合复杂度分数 [0, 1]
        complexity = 0.5 * normalize(point_density) + 0.5 * normalize(view_angle_range)
        return complexity
    
    def compute_margin(self, scene_complexity, dist_to_camera):
        """
        动态计算 margin
        """
        # 复杂场景：使用更小的 margin（保守剔除）
        # 简单场景：使用更大的 margin（激进剔除）
        complexity_factor = 1.0 - 0.5 * scene_complexity  # [0.5, 1.0]
        
        # 远处：更大的 margin
        # 近处：更小的 margin
        distance_factor = 1.0 + 0.1 * dist_to_camera
        
        margin = self.base_margin * complexity_factor * distance_factor
        return margin
```

class FeedbackBasedMarginController:
    def __init__(self, base_margin=0.1, target_culling_rate=0.3):
        self.base_margin = base_margin
        self.target_culling_rate = target_culling_rate
        self.culling_history = []
    
    def update_margin(self, current_culling_rate):
        """
        根据剔除率反馈调整 margin
        """
        self.culling_history.append(current_culling_rate)
        
        # 使用最近 10 帧的平均剔除率
        if len(self.culling_history) > 10:
            avg_culling_rate = np.mean(self.culling_history[-10:])
        else:
            avg_culling_rate = current_culling_rate
        
        # PID 控制思路
        error = self.target_culling_rate - avg_culling_rate
        
        # 剔除率过低 → 增加 margin（更激进）
        # 剔除率过高 → 减小 margin（更保守）
        adjustment = 0.01 * error  # 学习率
        
        new_margin = self.base_margin + adjustment
        new_margin = np.clip(new_margin, 0.05, 0.30)  # 限制范围
        
        self.base_margin = new_margin
        return new_margin
```
