```python
import torch
from typing import Dict, List
from dataclasses import dataclass

@dataclass
class ConvergenceState:
    """收敛状态"""
    iteration: int
    loss: float
    gradient_norm: float
    gaussian_count: int
    is_converged: bool

class ConvergenceAnalyzer:
    """
    收敛性分析器
    
    实现：
    - 收敛性检测
    - 学习率自适应调整
    - 稳定性监控
    """
    
    def __init__(self, window_size: int = 100):
        self.window_size = window_size
        self.loss_history: List[float] = []
        self.gradient_history: List[float] = []
        self.convergence_threshold = 1e-4
    
    def update(self, loss: float, gradient_norm: float, gaussian_count: int):
        """更新分析器状态"""
        self.loss_history.append(loss)
        self.gradient_history.append(gradient_norm)
        
        # 保持窗口大小
        if len(self.loss_history) > self.window_size:
            self.loss_history.pop(0)
            self.gradient_history.pop(0)
    
    def check_convergence(self) -> ConvergenceState:
        """
        检查是否收敛
        
        Returns:
            ConvergenceState 对象
        """
        if len(self.loss_history) < self.window_size:
            return ConvergenceState(
                iteration=len(self.loss_history),
                loss=self.loss_history[-1] if self.loss_history else float('inf'),
                gradient_norm=self.gradient_history[-1] if self.gradient_history else float('inf'),
                gaussian_count=0,
                is_converged=False
            )
        
        # 计算 loss 变化率
        recent_losses = self.loss_history[-10:]
        loss_change = (recent_losses[-1] - recent_losses[0]) / (recent_losses[0] + 1e-8)
        
        # 计算梯度范数
        avg_gradient = sum(self.gradient_history[-10:]) / 10
        
        is_converged = (
            abs(loss_change) < self.convergence_threshold and
            avg_gradient < self.convergence_threshold
        )
        
        return ConvergenceState(
            iteration=len(self.loss_history),
            loss=self.loss_history[-1],
            gradient_norm=avg_gradient,
            gaussian_count=0,  # TODO: 传入
            is_converged=is_converged
        )
    
    def get_optimal_learning_rate(self, base_lr: float) -> float:
        """
        基于收敛性分析推荐学习率
        
        Returns:
            推荐的学习率
        """
        if len(self.loss_history) < 10:
            return base_lr
        
        # 检测振荡
        recent_losses = self.loss_history[-10:]
        oscillation = max(recent_losses) - min(recent_losses)
        
        if oscillation > 0.1:
            # 振荡大，降低学习率
            return base_lr * 0.5
        elif oscillation < 0.01:
            # 稳定，可以提高学习率
            return base_lr * 1.1
        
        return base_lr
```

```python
class OptimalityBounds:
    """
    最优性边界计算
    
    实现理论推导的各种边界：
    - 高斯数量下界
    - 逼近误差上界
    - 计算复杂度边界
    """
    
    @staticmethod
    def compute_min_gaussians(target_psnr: float, scene_type: str = 'indoor') -> int:
        """
        计算达到目标 PSNR 所需的最少高斯数量
        
        基于信息论推导的下界
        
        Args:
            target_psnr: 目标 PSNR (dB)
            scene_type: 场景类型 ('indoor', 'outdoor', 'urban')
        
        Returns:
            最少高斯数量
        """
        # 基于经验公式的理论推导
        # PSNR = a * log(N) + b
        # => N = exp((PSNR - b) / a)
        
        coefficients = {
            'indoor': {'a': 5.0, 'b': 10.0},
            'outdoor': {'a': 6.0, 'b': 12.0},
            'urban': {'a': 7.0, 'b': 15.0}
        }
        
        coeff = coefficients.get(scene_type, coefficients['indoor'])
        min_count = int(torch.exp((target_psnr - coeff['b']) / coeff['a']))
        
        return max(min_count, 10_000)
    
    @staticmethod
    def compute_rate_distortion_bound(gaussians: Dict[str, torch.Tensor],

        计算率失真边界
```
